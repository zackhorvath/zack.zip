{{- /* Page-bundle images get width/height, a WebP srcset, and a larger rendition for the lightbox. site.js groups consecutive images into captioned figures and justified galleries. */ -}} {{- $image := .Page.Resources.GetMatch (printf "%s" (.Destination | safeURL)) -}} {{- $src := .Destination | relURL | safeURL -}} {{- $alt := .PlainText -}} {{- $attrs := dict -}} {{- if and $image (ne $image.MediaType.SubType "svg") (ne $image.MediaType.SubType "gif") -}} {{- $src = $image.RelPermalink -}} {{- $srcset := slice -}} {{- $last := false -}} {{- range site.Params.imageProcessing.content.widths -}} {{- if lt . $image.Width -}} {{- $r := $image.Resize (printf "%dx webp q85" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $r.Width) -}} {{- $last = $r -}} {{- end -}} {{- end -}} {{- $full := $image -}} {{- if gt $image.Width 2000 }}{{ $full = $image.Resize "2000x webp q85" }}{{ end -}} {{- if $last }}{{ $src = $last.RelPermalink }}{{ end -}} {{- $srcset = $srcset | append (printf "%s %dw" $full.RelPermalink $full.Width) -}} {{- $attrs = dict "width" $image.Width "height" $image.Height "srcset" (delimit $srcset ", ") "full" $full.RelPermalink -}} {{- end -}} {{- /**/ -}}
